# DOG5 SDK runtime dependencies.

# mujoco is NOT optional.  It is the simulator, and it is also the authority
# the kinematics, the statics and the robot's mass are gated against.
numpy>=1.24

# Only needed to talk to the real robot.  `import dog5_sdk` and Dog5Sim do not
# touch it -- see dog5_sdk/motor/__init__.py -- so the simulator runs on a
# laptop with no CAN adapter even if this is missing.
python-can>=4.0
